Используется в шаблоне: "/pkforum.m3ra.ru/site/layouts/speakers/list.html"

title: "Артём Сверкунов"
loc: "Россия"

telegram: "Artemych_33"

  - title: VK
    link: https://vk.com/sverkunov_av
  - title: Сайт
    link: https://www.neufeld.ru/

  - title: Здравница
